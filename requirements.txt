# --- web surface --------------------------------------------------------
# One process serves the API and both interfaces. No build step, no second
# command: the submission rules allow exactly one.
fastapi>=0.115
uvicorn>=0.30
python-multipart>=0.0.9

# --- configuration and contracts ---------------------------------------
pydantic>=2.9
python-dotenv>=1.0

# --- the only thing that goes over the network --------------------------
# Gemini, for both the LLM calls and the embeddings (gemini-embedding-001).
# Everything else - retrieval, storage, the rule engine - is local.
google-genai>=1.0

# --- local retrieval ----------------------------------------------------
# Cosine similarity over a few hundred committed vectors. No FAISS, no Chroma:
# at this corpus size an index costs more startup time than it saves, and it is
# one more dependency that can fail on a machine we do not control.
numpy>=1.26

# --- tool protocol ------------------------------------------------------
# The same tool layer the application uses is published over MCP stdio, so an
# external client can drive VITA with the identical implementation. Local
# transport only; nothing here goes over the network.
mcp>=1.2

# --- patient memory -----------------------------------------------------
# MemPalace holds what VITA remembers between visits: previous presentations,
# whether they resolved, durable patient facts, clinician notes. Configured
# with the sqlite_exact backend (local file, sqlite3 + numpy) and with
# gemini-embedding-001 injected as its embedder, so its default ONNX weights
# are never downloaded and nothing but Gemini crosses the network.
mempalace==3.6.0
